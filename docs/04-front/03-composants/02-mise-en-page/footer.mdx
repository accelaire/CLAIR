import { Footer } from "@clair/ui";

Pied de page. Chaque colonne est une navigation nommée par son titre ; les sources des données sont des liens externes annoncés comme tels. [Toutes les variantes dans l'atelier Storybook](https://storybook.clair.vote/?path=/docs/composants-mise-en-page-footer--docs)

## Import

```tsx
import { Footer } from "@clair/ui";
```

## Utilisation de base

<Preview surface="page">
  <Footer
    columns={[
      { title: "Explorer", links: [{ label: "Députés", href: "/deputes" }] },
      { title: "Légal", links: [{ label: "CGU", href: "/cgu" }] },
    ]}
    year={2026}
  />
</Preview>

```tsx
<Footer
  columns={[
    { title: "Explorer", links: [{ label: "Députés", href: "/deputes" }] },
    { title: "Légal", links: [{ label: "CGU", href: "/cgu" }] },
  ]}
/>
```
